#Bu kodda  fwi hesaplamalarının her bir istasyon için daha kolay hesaplanabileceği process_fwi fonksiyonu tanımlanır.

#Yüklenmesi gereken package'lar 
library(cffdrs)
library(readxl)
library(openxlsx)

# Fonksiyonu tanımlama
process_fwi <- function(input_file, output_file, init_values) {
  # Veriyi okuma
  fire <- read_excel(input_file)
  
  # Fwi hesaplama
  fwi_result <- fwi(fire, init = init_values)
  
  # Fwi verisini dataframe'e dönüştürme
  df <- data.frame(FWI = fwi_result)
  
  # Veriyi Excel'e dönüştürme
  write.xlsx(df, output_file)
}
